{{/* Returns a post's frontmatter date as time. Handles legacy "YYYY-MM-DDTHHZ" strings Hugo can't parse. */}} {{ $date := .Date }} {{ $rawDate := printf "%v" .Params.date }} {{ if gt (len (findRE "^\\d{4}-\\d{2}-\\d{2}T\\d{2}Z$" $rawDate)) 0 }} {{ $date = time.AsTime (replaceRE "^([0-9]{4}-[0-9]{2}-[0-9]{2})T([0-9]{2})Z$" "${1}T${2}:00:00Z" $rawDate) }} {{ else if .Params.date }} {{ $date = time.AsTime .Params.date }} {{ end }} {{ return $date }}
